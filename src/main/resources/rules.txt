# =============================================================================
# Ecosystem Simulation — User-Defined Rules
# =============================================================================
# Format:  RuleName | TargetType | Condition | Action
#
# TargetType supports the full class hierarchy:
#   Entity, Organism, Animal, Predator, Herbivore, Plant
#
# Condition fields:   energy, age, x, y
# Condition operators: <  >  <=  >=  ==  !=
#
# Action keywords:
#   die                 — entity is removed from the simulation
#   move                — animal moves one step randomly
#   flee                — herbivore retreats (falls back to random move)
#   grow                — plant gains energy
#   reproduce           — entity attempts to reproduce
#   energy -N           — organism loses N energy units  (NEW)
#   energy +N           — organism gains N energy units  (NEW)
#
# Lines starting with # are comments and are ignored.
# =============================================================================

# --- Predator rules ---

# Predators die if energy is critically low
StarvingPredator | Predator | energy < 10 | die

# Cold penalty: simulates harsh weather draining predator energy faster
# (demonstrates the new  energy -N  action syntax)
ColdPenalty | Predator | age > 30 | energy -5

# Desperate predators keep moving in search of food
DesperatePredator | Predator | energy < 20 | move

# --- Herbivore rules ---

# Old herbivores flee more aggressively (triggered by rule engine)
OldHerbivore | Herbivore | age > 50 | flee

# Well-fed herbivores gain a small energy bonus (abundant-food bonus)
WellFedBonus | Herbivore | energy >= 70 | energy +5

# --- Plant rules ---

# Plants with enough stored energy grow faster
FastPlantGrowth | Plant | energy > 30 | grow

# --- Cross-species rules (use base types for polymorphic matching) ---

# Any organism that exceeds maximum lifespan dies of old age
MaximumAge | Organism | age > 100 | die

# Any animal that reaches very old age receives a moderate energy penalty
AgingPenalty | Animal | age > 60 | energy -3

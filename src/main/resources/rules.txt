# =============================================================================
# Ecosystem Simulation -- Bundled Default Rules (read-only seed)
# =============================================================================
# This file is packaged on the classpath and used only once, to bootstrap
# config/rules.txt the first time the application runs. After that, every
# load/save/edit touches config/rules.txt only -- this file is never written to.
#
# Format:  RuleName | TargetType | condition | actionList
#
# TargetType: Entity, Organism, Animal, Predator, Herbivore, Plant
#
# condition   := boolean expression with AND / OR / NOT / parentheses / comparisons
# actionList  := one or more actions separated by commas
# action      := <writable-attribute> (= | += | -= | *= | /=) <numeric-expr>
#              | <domain-command>
#
# Readable references : energy, age, x, y, speed, attackPower, defensePower,
#                        growthRate, env.foodLevel, env.temperature, env.width,
#                        env.height, stat.predatorPopulation,
#                        stat.herbivorePopulation, stat.plantPopulation
# Writable attributes  : energy, speed, attackPower, defensePower, growthRate
# Domain commands      : die, move, flee, grow, reproduce
#
# Lines starting with # are comments and are ignored.
# =============================================================================

# --- Predator rules ---

StarvingPredator | Predator | energy < 10 | die

# Cold penalty: harsh temperature drains predator energy faster
ColdPenalty | Predator | age > 30 AND env.temperature < 10 | energy -= 5

DesperatePredator | Predator | energy < 20 | move

# --- Herbivore rules ---

OldHerbivore | Herbivore | age > 50 | flee

WellFedBonus | Herbivore | energy >= 70 | energy += 5

# Compound temperature rule: cold and hungry herbivores lose energy and slow down
ColdStress | Herbivore | env.temperature < 5 AND energy < 40 | energy -= 8, speed *= 0.8

# --- Plant rules ---

FastPlantGrowth | Plant | energy > 30 | grow

# --- Cross-species rules (target a superclass to match every subtype) ---

MaximumAge | Organism | age > 100 | die

AgingPenalty | Animal | age > 60 | energy -= 3

# Population-pressure rule referencing simulation statistics: demonstrates a
# rule that could never be expressed by the old preset-dropdown rule builder
# (no AND, no statistics reference, no non-energy attribute mutation existed).
CrowdedForest | Plant | stat.plantPopulation > 100 OR env.foodLevel < 20 | growthRate *= 0.5
